ingredients:
  - amount: "1 can"
    item: "Black beans, rinsed"
  - amount: "1 can"
    item: "Chickpeas, rinsed"
  - amount: "1 cup"
    item: "Cherry tomatoes, halved"
  - amount: "1"
    item: "Avocado, diced"
  - amount: "½ cup"
    item: "Red cabbage, shredded"
  - amount: "1"
    item: "Yellow bell pepper, diced"
  - amount: "½ cup"
    item: "Sweet potato, roasted & cubed"
  - amount: "2 tbsp"
    item: "Olive oil"
  - amount: "1"
    item: "Lime, juiced"
  - amount: "To taste"
    item: "Salt & cracked pepper"
steps:
  - title: "Roast the sweet potato"
    description: "Toss cubed sweet potato with a little olive oil and salt. Roast at 400°F for 20 minutes, until edges caramelize. Let cool slightly."
  - title: "Prep the vegetables"
    description: "While the sweet potato roasts, halve the tomatoes, dice the bell pepper and avocado, and shred the cabbage."
  - title: "Combine"
    description: "In a large bowl, combine the beans, chickpeas, tomatoes, cabbage, bell pepper, and roasted sweet potato."
  - title: "Dress and finish"
    description: "Whisk the olive oil and lime juice together, pour over the salad, and toss gently. Fold in the avocado last so it stays intact. Season to taste."
note: "This keeps well in the fridge for up to 3 days — just hold the avocado back and add it fresh right before serving so it doesn't brown. Great scooped with grain-free crackers or spooned over greens for a heartier lunch."
